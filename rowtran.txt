import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.print("Enter message: ");
        String s = sc.nextLine().replace(" ", "").toUpperCase();

        int n = 4;
        while (s.length() % n != 0) s += "X";

        int r = s.length() / n, k = 0;
        char[][] a = new char[r][n];

        for (int i = 0; i < r; i++)
            for (int j = 0; j < n; j++)
                a[i][j] = s.charAt(k++);

        String enc = "", dec = "";

        for (int j = 0; j < n; j++)
            for (int i = 0; i < r; i++)
                enc += a[i][j];

        k = 0;
        for (int j = 0; j < n; j++)
            for (int i = 0; i < r; i++)
                a[i][j] = enc.charAt(k++);

        for (int i = 0; i < r; i++)
            for (int j = 0; j < n; j++)
                dec += a[i][j];

        System.out.println("Encrypted: " + enc);
        System.out.println("Decrypted: " + dec.replaceAll("X+$", ""));
    }
}
